cmake_minimum_required(VERSION 3.16)
project(DrivOnix LANGUAGES C CXX)

include(GNUInstallDirs)
option(DRIVONIX_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The graphical emulator currently supports Linux and Windows")
endif()

# Use native CSFML 2 libraries. The repository's sfml/*.a and *.dll files are
# legacy 32-bit Windows binaries and must never enter a Linux build.
set(CSFML_ROOT "" CACHE PATH "CSFML 2 installation prefix")
find_path(CSFML_INCLUDE_DIR SFML/Graphics.h HINTS ${CSFML_ROOT} ENV CSFML_ROOT PATH_SUFFIXES include)
foreach(component graphics window system audio)
    find_library(CSFML_${component}_LIBRARY NAMES csfml-${component}
        HINTS ${CSFML_ROOT} ENV CSFML_ROOT PATH_SUFFIXES lib lib64)
endforeach()
include(FindPackageHandleStandardArgs)
find_package_handle_standard_args(CSFML REQUIRED_VARS CSFML_INCLUDE_DIR
    CSFML_graphics_LIBRARY CSFML_window_LIBRARY CSFML_system_LIBRARY CSFML_audio_LIBRARY)
if(NOT CSFML_FOUND)
    message(FATAL_ERROR "Install the CSFML 2 development package or set CSFML_ROOT to its installation prefix")
endif()
file(STRINGS "${CSFML_INCLUDE_DIR}/SFML/Config.h" csfml_minor REGEX "^#define CSFML_VERSION_MINOR")
file(STRINGS "${CSFML_INCLUDE_DIR}/SFML/Config.h" csfml_major REGEX "^#define CSFML_VERSION_MAJOR")
if(NOT csfml_major MATCHES "CSFML_VERSION_MAJOR +2$" OR NOT csfml_minor MATCHES "CSFML_VERSION_MINOR +[6-9]$")
    message(FATAL_ERROR "DrivOnix requires CSFML 2.6 or later in the 2.x series (the CSFML 3 API is incompatible)")
endif()

set(DRIVONIX_MACHINE_SOURCES
    genesis/genesis.cpp genesis/romloader/romloader.cpp genesis/gamepad/gamepad.cpp genesis/cartridge/cartridge.cpp genesis/cartridge/eeprom.cpp
    bits/bitsutils.cpp m68k/m68k.cpp ym7101/ym7101.cpp cathoderaytube/crt.cpp
    z80/z80.cpp z80/core.c audio/audio.cpp thirdparty/ymfm/ymfm_opn.cpp
    thirdparty/ymfm/ymfm_ssg.cpp thirdparty/ymfm/ymfm_adpcm.cpp platform/settings.cpp
    platform/controller.cpp frontend/input.cpp)
add_library(drivonix_content STATIC platform/archive.cpp platform/process.cpp platform/storage.cpp library/rominfo.cpp)
target_compile_features(drivonix_content PUBLIC cxx_std_14)
set_target_properties(drivonix_content PROPERTIES CXX_EXTENSIONS OFF)
add_library(drivonix_sqlite STATIC thirdparty/sqlite/sqlite3.c)
target_compile_definitions(drivonix_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_DEFAULT_FILE_PERMISSIONS=0600)
target_include_directories(drivonix_sqlite PUBLIC thirdparty/sqlite)
add_library(drivonix_library STATIC library/library.cpp)
target_compile_features(drivonix_library PUBLIC cxx_std_14)
target_link_libraries(drivonix_library PUBLIC drivonix_content drivonix_sqlite)
add_library(drivonix_metadata STATIC metadata/catalog.cpp metadata/http.cpp metadata/service.cpp)
target_compile_features(drivonix_metadata PUBLIC cxx_std_14)
target_link_libraries(drivonix_metadata PUBLIC drivonix_content)
if(WIN32)
    target_link_libraries(drivonix_metadata PRIVATE winhttp)
    target_link_libraries(drivonix_content PUBLIC shell32 ole32 uuid)
else()
    find_package(CURL REQUIRED)
    target_link_libraries(drivonix_metadata PRIVATE CURL::libcurl)
endif()
add_library(drivonix_console STATIC ${DRIVONIX_MACHINE_SOURCES} platform/platform.cpp platform/audiooutput.cpp)
target_compile_features(drivonix_console PUBLIC cxx_std_14)
set_target_properties(drivonix_console PROPERTIES CXX_EXTENSIONS OFF C_STANDARD 11)
target_include_directories(drivonix_console SYSTEM PUBLIC ${CSFML_INCLUDE_DIR})
target_link_libraries(drivonix_console PUBLIC ${CSFML_graphics_LIBRARY} ${CSFML_window_LIBRARY} ${CSFML_system_LIBRARY} ${CSFML_audio_LIBRARY})
find_package(Threads REQUIRED)
target_link_libraries(drivonix_console PUBLIC Threads::Threads)
target_link_libraries(drivonix_content PUBLIC Threads::Threads)
target_link_libraries(drivonix_console PUBLIC drivonix_content)
target_link_libraries(drivonix_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
target_compile_definitions(drivonix_console PRIVATE DRIVONIX_INSTALL_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/drivonix")

if(WIN32)
    target_link_libraries(drivonix_console PUBLIC comdlg32 shell32 ole32 uuid)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
    target_link_libraries(drivonix_console PRIVATE PkgConfig::GTK3)
endif()
if(MSVC)
    target_compile_options(drivonix_console PRIVATE /utf-8 /bigobj)
    foreach(target drivonix_content drivonix_library drivonix_metadata)
        target_compile_options(${target} PRIVATE /utf-8)
    endforeach()
endif()

add_library(drivonix_session STATIC frontend/session.cpp)

target_link_libraries(drivonix_session PUBLIC drivonix_console drivonix_content)

    target_link_options(drivonix_console PUBLIC -fsanitize=address,undefined)
    foreach(target drivonix_content drivonix_library drivonix_metadata drivonix_sqlite)

    endforeach()

add_custom_command(TARGET DrivOnix POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:DrivOnix>/bios"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/bios/drivonix.bios" "$<TARGET_FILE_DIR:DrivOnix>/bios/drivonix.bios"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:DrivOnix>/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/bios/res/biosbackground.bmp" "$<TARGET_FILE_DIR:DrivOnix>/assets/branding.bmp"
    VERBATIM)

install(TARGETS DrivOnix RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES bios/drivonix.bios DESTINATION ${CMAKE_INSTALL_DATADIR}/drivonix/bios)
install(FILES config/drivonix.ini DESTINATION ${CMAKE_INSTALL_DATADIR}/drivonix/config)
install(DIRECTORY assets/ DESTINATION ${CMAKE_INSTALL_DATADIR}/drivonix/assets PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES bios/res/biosbackground.bmp DESTINATION ${CMAKE_INSTALL_DATADIR}/drivonix/assets RENAME branding.bmp)
install(DIRECTORY thirdparty/ DESTINATION ${CMAKE_INSTALL_DATADIR}/drivonix/third-party)
